时间复杂度的优化：O(N) => O(1)

简单来说就是针对调用频率很高的重复性求和类的操作，进行一次性的计算，把上下左右等方向的和或者积等提前算好维护起来，需要用的时候直接取值即可

![图解](/img/algorithm/1.jpeg)

```js
var NumArray = (nums) => {
  let preSum = new Array(nums.length + 1).fill(0);

  preSum[0] = 0;

  for (let i = 0; i < nums.length; i++) {
    preSum[i + 1] = preSum[i] + nums[i];
  }

  this.preSum = preSum;
};

NumArray.prototype.sumRange = function (left, right) {
  return this.preSum[right + 1] - this.preSum[left];
};
```

```js
var NumMatrix = (matrix) => {
  let m = matrix.length;
  let n = matrix[0].length;
  // preSum[i][j] 表示 matrix[0..i-1][0..j-1] 的元素和
  let preSum = new Array(m + 1).fill(0).map(() => new Array(n + 1).fill(0));

  for (let i = 0; i < m; i++) {
    for (let j = 0; j < n; j++) {
      // 计算每个矩阵[0,0,i,j]的和
      preSum[i + 1][j + 1] = preSum[i][j + 1] + preSum[i + 1][j] - preSum[i][j] + matrix[i][j];
    }
  }

  this.preSum = preSum;
};

NumMatrix.prototype.sumRegion = function (row1, col1, row2, col2) {
  // 计算子矩阵[row1,col1,row2,col2]的和
  // 目标矩阵之和由四个相邻矩阵运算组成
  return (
    this.preSum[row2 + 1][col2 + 1] -
    this.preSum[row1][col2 + 1] -
    this.preSum[row2 + 1][col1] +
    this.preSum[row1][col1]
  );
};
```
